\section{هدف اسپرینت}

پس از اسپرینت اول که در آن زیرساخت احراز هویت پایه‌ریزی شد، اسپرینت دوم بر توسعه
کامل قابلیت‌های اصلی سیستم تمرکز داشت. در پایان این اسپرینت سیستم باید یک
\lr{MVP} (حداقل محصول قابل ارائه) کامل داشته باشد.

\subsection{اهداف اصلی}
\begin{itemize}[rightmargin=0.5cm]
  \item پیاده‌سازی \lr{CRUD} کامل رستوران (ایجاد، مشاهده، ویرایش، تأیید)
  \item مدیریت منو و میزهای رستوران
  \item سیستم رزرو کامل (ایجاد، تأیید، رد، لغو)
  \item قابلیت علاقه‌مندی‌ها
  \item داشبورد برای هر سه نقش
  \item اتصال تمام صفحات فرانت‌اند به \lr{API} واقعی
\end{itemize}

\section{بک‌اند: سیستم رزرو}

\subsection{قانون کسب‌وکار}
\begin{itemize}[rightmargin=0.5cm]
  \item مدت رزرو ثابت ۲ ساعته است: \lr{EndTime = StartTime + 2 hours}
  \item میز انتخاب‌شده باید در بازه زمانی درخواستی در دسترس باشد
  \item رزرو جدید با وضعیت \lr{Pending} ثبت می‌شود
  \item مشتری می‌تواند تعداد مهمانان متفاوت از ظرفیت میز (تا سقف) داشته باشد
\end{itemize}

\subsection{چرخه عمر رزرو}
\begin{enumerate}[rightmargin=0.5cm]
  \item \textbf{Pending}: مشتری رزرو ثبت کرده، در انتظار تأیید مدیر رستوران
  \item \textbf{Confirmed}: مدیر رستوران رزرو را تأیید کرد
  \item \textbf{Rejected}: مدیر رستوران رزرو را رد کرد
  \item \textbf{Cancelled}: مشتری یا مدیر رستوران رزرو را لغو کرد
\end{enumerate}

\subsection{فرم رزرو (ReservationForm)}

صفحه رزرو میز شامل مراحل زیر است:
\begin{enumerate}[rightmargin=0.5cm]
  \item انتخاب تاریخ با تقویم شمسی
  \item انتخاب ساعت شروع از میان ساعات کاری رستوران
  \item وارد کردن تعداد مهمانان
  \item نمایش میزهای در دسترس و انتخاب میز
  \item وارد کردن نام و شماره تماس (در صورت تفاوت با حساب کاربری)
  \item تأیید و ارسال رزرو
\end{enumerate}

\subsection{داشبورد مدیر رستوران (ManagerDashboard)}

داشبورد مدیر رستوران شامل:
\begin{itemize}[rightmargin=0.5cm]
  \item آمار کلی: تعداد رزروها، تأییدشده‌ها، در انتظار، لغوشده‌ها
  \item لیست رزروهای امروز با امکان تأیید/رد
  \item نمودار وضعیت رزروها با \lr{Recharts}
  \item منوی ناوبری به صفحات مدیریت منو، میز و ساعات کاری
\end{itemize}

\subsection{ثبت رستوران چندمرحله‌ای (RestaurantOnboarding)}

صفحه ثبت رستوران (\lr{/onboarding}) یک فرم چندمرحله‌ای است با مراحل:
\begin{enumerate}[rightmargin=0.5cm]
  \item اطلاعات پایه: نام، توضیح
  \item موقعیت: آدرس، استان/شهر
  \item نوع غذا (\lr{FoodType})
  \item ساعات کاری اولیه
  \item آپلود تصویر اصلی رستوران
  \item مرور و تأیید نهایی
\end{enumerate}

\section{بازبینی و گذر اسپرینت}

\subsection{Sprint Review}
\begin{itemize}[rightmargin=0.5cm]
  \item تمام ۳۵ آیتم بک‌لاگ با موفقیت انجام شدند
  \item تمام \lr{Endpoint}های بک‌اند از طریق \lr{Swagger} تست شدند
  \item صفحات فرانت‌اند به \lr{API} واقعی وصل شدند
  \item جریان کامل رزرو (از جستجو تا تأیید) کار می‌کند
\end{itemize}

\subsection{Sprint Retrospective}

\textbf{چه چیزی خوب بود:}
\begin{itemize}[rightmargin=0.5cm]
  \item معماری \lr{CQRS} کدها را سازماندهی و تست‌پذیر نگه داشت
  \item پیاده‌سازی موازی بک‌اند و فرانت‌اند کارایی را افزایش داد
  \item استفاده از \lr{FluentValidation} خطاهای ورودی را به خوبی مدیریت کرد
\end{itemize}

\textbf{چه چیزی می‌توانست بهتر باشد:}
\begin{itemize}[rightmargin=0.5cm]
  \item اتصال تقویم شمسی در فرم رزرو نیاز به کار بیشتری داشت
  \item \lr{TableManagement} به اسپرینت سوم منتقل شد
\end{itemize}

\textbf{اقدامات بهبود:}
\begin{itemize}[rightmargin=0.5cm]
  \item اضافه کردن تست‌های یکپارچه‌سازی برای سیستم رزرو
  \item بهبود مدیریت خطا در سمت فرانت‌اند
\end{itemize}
